function render(){
  document.querySelectorAll('.sec').forEach(function(s){s.classList.remove('on')});
  document.getElementById('s'+aW).classList.add('on');
  var q=document.getElementById('srch').value.toLowerCase().trim();
  var src=D[aW]||[];
  var rows=src.filter(function(w){
    if(aR!=='ALL'&&w.region!==aR)return false;
    if(aG!=='ALL'&&w.grade!==aG)return false;
    if(q){var h=[w.name,w.school,w.region,w.grade,w.piaa,w.results,w.rank].join(' ').toLowerCase();if(h.indexOf(q)<0)return false;}
    return true;
  });
  rows=rows.slice().sort(function(a,b){
    var av=a[sC],bv=b[sC];
    if(sC==='rank')return(rs(av)-rs(bv))*sDir;
    if(sC==='grade')return((+av||0)-(+bv||0))*sDir;
    av=(av||'').toLowerCase();bv=(bv||'').toLowerCase();
    return(av<bv?-1:av>bv?1:0)*sDir;
  });
  var tb=document.getElementById('b'+aW);
  if(!rows.length){tb.innerHTML='<tr><td colspan="8" class="empty">No wrestlers match.</td></tr>';document.getElementById('cnt').textContent='0 wrestlers';return;}
  var wc=aW;
  tb.innerHTML=rows.map(function(w,i){
    var nr=w.rank==='NR'||w.rank==='INJ';
    var idx=src.indexOf(w);
    return '<tr class="'+(i%2===0?'a':'b')+'">'
      +'<td class="rk'+(nr?' nr':'')+'">'+w.rank+'</td>'
      +'<td class="rg">'+(w.region||'--')+'</td>'

      +'<td class="rs">'+hl(w.results)+'</td>'
      +'</tr>';
  }).join('');
  document.getElementById('cnt').textContent=rows.length+' wrestlers';
}
